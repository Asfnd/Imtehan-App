'use client'

import { useState, useEffect, useMemo } from 'react'
import { createClientComponentClient } from '@supabase/auth-helpers-nextjs'
import { FileText, Calendar, Search, X } from 'lucide-react'
import { useRouter } from 'next/navigation'
import Link from 'next/link'
import FeedbackButton from '@/components/FeedbackButton'
import ProtectedContent from '@/components/security/ProtectedContent'
import DevToolsWarning from '@/components/security/DevToolsWarning'
import { usageTracker } from '@/lib/usageTracker'
import SignUpModal from '@/components/SignUpModal'

interface Paper {
  id: number
  subject: string
  year: number | null
  filename: string
  storage_path: string
  file_size: number
  is_available: boolean
  download_count: number
}

export default function PastPapersPage() {
  const router = useRouter()
  const [papers, setPapers] = useState<Record<string, number[]>>({})
  const [loading, setLoading] = useState(true)
  const [selectedSubject, setSelectedSubject] = useState<string | null>(null)
  const [searchQuery, setSearchQuery] = useState('')
  const [showSignUpModal, setShowSignUpModal] = useState(false)
  const [user, setUser] = useState<any>(null)
  const supabase = createClientComponentClient()

  // Check auth status
  useEffect(() => {
    const checkUser = async () => {
      const { data: { user } } = await supabase.auth.getUser()
      setUser(user)
    }
    checkUser()
  }, [])

  useEffect(() => {
    loadPapers()
  }, [])

  async function loadPapers() {
    try {
      console.log('Loading papers from past_papers table...')
      
      // Fetch all papers from the database
      const { data, error } = await supabase
        .from('past_papers')
        .select('*')
        .eq('is_available', true)
        .order('year', { ascending: false })

      console.log('Database response:', { data, error })

      if (error) {
        console.error('Database error:', error)
        throw error
      }

      if (!data || data.length === 0) {
        console.warn('No papers found in database')
        setPapers({})
        return
      }

      console.log(`Found ${data.length} papers`)

      // Organize by subject
      const organized: Record<string, number[]> = {}
      
      data.forEach((paper: Paper) => {
        if (paper.year) {
          if (!organized[paper.subject]) {
            organized[paper.subject] = []
          }
          if (!organized[paper.subject].includes(paper.year)) {
            organized[paper.subject].push(paper.year)
          }
        }
      })

      // Sort years for each subject (newest first)
      Object.keys(organized).forEach(subject => {
        organized[subject].sort((a, b) => b - a)
      })

      console.log('Organized papers:', organized)
      setPapers(organized)
    } catch (error: unknown) {
      console.error('Error loading papers:', error)
      const errorMessage = error instanceof Error ? error.message : 'Unknown error'
      alert(`Error loading papers: ${errorMessage}. Check console for details.`)
    } finally {
      setLoading(false)
    }
  }

  const formatSubjectName = (slug: string) => {
    return slug
      .split('-')
      .map(word => word.charAt(0).toUpperCase() + word.slice(1))
      .join(' ')
  }

  const subjects = useMemo(() => Object.keys(papers).sort(), [papers])

  const filteredSubjects = useMemo(() => {
    if (!searchQuery) return subjects
    return subjects.filter(subject =>
      formatSubjectName(subject).toLowerCase().includes(searchQuery.toLowerCase())
    )
  }, [subjects, searchQuery])

  if (loading) {
    return (
      <div className="min-h-screen bg-gradient-to-br from-slate-50 to-green-50 flex items-center justify-center">
        <div className="text-center">
          <div className="w-12 h-12 border-4 border-green-200 border-t-green-600 rounded-full animate-spin mx-auto"></div>
          <p className="mt-4 text-gray-600 font-medium">Loading papers...</p>
        </div>
      </div>
    )
  }

  return (
    <>
      <DevToolsWarning />
      <ProtectedContent>
        <div className="min-h-screen bg-gradient-to-br from-green-50 via-emerald-50 to-teal-50 p-4">
          <div className="max-w-7xl mx-auto">
            {/* Modern Header with Green Theme */}
            <div className="flex items-center justify-between mb-6">
              <button
                onClick={() => router.back()}
                className="flex items-center gap-2 px-4 py-2 bg-white/80 hover:bg-white text-gray-700 hover:text-gray-900 rounded-xl transition-all shadow-sm hover:shadow-md font-medium"
              >
                ← Back
              </button>
              <div className="text-center">
                <h1 className="text-2xl md:text-3xl font-bold bg-gradient-to-r from-green-600 via-emerald-600 to-teal-600 bg-clip-text text-transparent">
                  CSS Past Papers
                </h1>
                <p className="text-sm text-gray-600 mt-1">Browse by subject & year</p>
              </div>
              <div className="w-24"></div>
            </div>

            {/* Ultra Modern Search Bar with Green Gradient */}
            <div className="mb-6">
              <div className="relative group">
                <div className="absolute -inset-0.5 bg-gradient-to-r from-green-400 via-emerald-400 to-teal-400 rounded-2xl opacity-20 group-hover:opacity-30 blur transition-opacity"></div>
                <div className="relative">
                  <Search className="absolute left-5 top-1/2 -translate-y-1/2 w-5 h-5 text-green-500 group-hover:text-green-600 transition-colors" />
                  <input
                    type="text"
                    placeholder="Search subjects... (e.g., International Relations)"
                    value={searchQuery}
                    onChange={(e) => setSearchQuery(e.target.value)}
                    className="w-full pl-14 pr-14 py-4 bg-white border-2 border-gray-200 rounded-2xl text-gray-900 placeholder-gray-400 focus:border-green-400 focus:ring-4 focus:ring-green-100 focus:outline-none transition-all shadow-sm hover:shadow-md"
                  />
                  {searchQuery && (
                    <button
                      onClick={() => setSearchQuery('')}
                      className="absolute right-5 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 hover:bg-gray-100 rounded-full p-1 transition-colors"
                    >
                      <X className="w-5 h-5" />
                    </button>
                  )}
                </div>
              </div>
            </div>

            {/* Green Stats Cards */}
            <div className="mb-6 grid grid-cols-3 gap-3">
              <div className="bg-gradient-to-br from-green-500 to-green-600 rounded-xl p-4 text-white shadow-lg">
                <div className="text-2xl font-bold">{subjects.length}</div>
                <div className="text-xs text-green-100">Subjects</div>
              </div>
              <div className="bg-gradient-to-br from-emerald-500 to-emerald-600 rounded-xl p-4 text-white shadow-lg">
                <div className="text-2xl font-bold">
                  {Object.values(papers).reduce((sum, years) => sum + years.length, 0)}
                </div>
                <div className="text-xs text-emerald-100">Years</div>
              </div>
              <div className="bg-gradient-to-br from-teal-500 to-teal-600 rounded-xl p-4 text-white shadow-lg">
                <div className="text-2xl font-bold">2000+</div>
                <div className="text-xs text-teal-100">Range</div>
              </div>
            </div>

            {/* Modern Subjects Grid with Green Action Buttons */}
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
              {filteredSubjects.map((subject, index) => {
                const colors = [
                  { bg: 'bg-green-50', border: 'border-green-200', text: 'text-green-600', btn: 'from-green-500 to-green-600', hover: 'hover:border-green-300' },
                  { bg: 'bg-emerald-50', border: 'border-emerald-200', text: 'text-emerald-600', btn: 'from-emerald-500 to-emerald-600', hover: 'hover:border-emerald-300' },
                  { bg: 'bg-teal-50', border: 'border-teal-200', text: 'text-teal-600', btn: 'from-teal-500 to-teal-600', hover: 'hover:border-teal-300' },
                  { bg: 'bg-lime-50', border: 'border-lime-200', text: 'text-lime-600', btn: 'from-lime-500 to-lime-600', hover: 'hover:border-lime-300' },
                  { bg: 'bg-cyan-50', border: 'border-cyan-200', text: 'text-cyan-600', btn: 'from-cyan-500 to-cyan-600', hover: 'hover:border-cyan-300' },
                  { bg: 'bg-green-50', border: 'border-green-200', text: 'text-green-700', btn: 'from-green-600 to-emerald-600', hover: 'hover:border-green-400' },
                ]
                const color = colors[index % colors.length]
                
                return (
                  <div
                    key={subject}
                    className={`bg-white rounded-2xl p-5 shadow-sm border-2 ${color.border} ${color.hover} transition-all`}
                  >
                    <div className="flex items-start justify-between mb-3">
                      <div className={`p-3 ${color.bg} rounded-xl`}>
                        <FileText className={`w-6 h-6 ${color.text}`} />
                      </div>
                      <span className={`text-xs font-bold ${color.text} ${color.bg} px-3 py-1.5 rounded-full`}>
                        {papers[subject].length} years
                      </span>
                    </div>
                    
                    <h3 className="font-bold text-lg text-gray-900 mb-2 line-clamp-2 leading-tight">
                      {formatSubjectName(subject)}
                    </h3>
                    
                    <div className="flex items-center gap-2 mb-4">
                      <Calendar className={`w-4 h-4 ${color.text}`} />
                      <p className="text-sm text-gray-600 font-medium">
                        {papers[subject][0]} - {papers[subject][papers[subject].length - 1]}
                      </p>
                    </div>
                    
                    <button
                      onClick={() => setSelectedSubject(subject)}
                      className={`w-full bg-gradient-to-r ${color.btn} text-white py-3 px-4 rounded-xl font-semibold hover:shadow-lg transition-all flex items-center justify-center gap-2 group`}
                    >
                      <span>View Papers</span>
                      <svg className="w-5 h-5 group-hover:translate-x-1 transition-transform" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                        <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 5l7 7-7 7" />
                      </svg>
                    </button>
                  </div>
                )
              })}
            </div>

            {filteredSubjects.length === 0 && (
              <div className="text-center py-16">
                <div className="inline-block p-4 bg-gradient-to-br from-green-100 to-emerald-100 rounded-full mb-4">
                  <Search className="w-12 h-12 text-green-500" />
                </div>
                <p className="text-gray-600 font-medium">No subjects found matching "{searchQuery}"</p>
                <button
                  onClick={() => setSearchQuery('')}
                  className="mt-4 px-6 py-2 bg-gradient-to-r from-green-500 to-emerald-500 text-white rounded-xl font-semibold hover:shadow-lg transition-all"
                >
                  Clear Search
                </button>
              </div>
            )}

            {/* Modern Year Modal with Green Theme */}
            {selectedSubject && (
              <div 
                className="fixed inset-0 bg-black/60 backdrop-blur-sm flex items-center justify-center p-4 z-50"
                onClick={() => setSelectedSubject(null)}
              >
                <div 
                  onClick={(e) => e.stopPropagation()}
                  className="bg-white rounded-3xl max-w-3xl w-full max-h-[85vh] overflow-hidden shadow-2xl"
                >
                  {/* Green Gradient Header */}
                  <div className="p-6 bg-gradient-to-r from-green-500 via-emerald-500 to-teal-500">
                    <div className="flex items-center justify-between">
                      <div>
                        <h2 className="text-2xl font-bold text-white mb-1">
                          {formatSubjectName(selectedSubject)}
                        </h2>
                        <p className="text-sm text-white/90">
                          {papers[selectedSubject].length} years available
                        </p>
                      </div>
                      <button
                        onClick={() => setSelectedSubject(null)}
                        className="text-white/80 hover:text-white hover:bg-white/20 w-10 h-10 flex items-center justify-center rounded-full transition-all"
                      >
                        <X className="w-6 h-6" />
                      </button>
                    </div>
                  </div>
                  
                  {/* Years Grid with Green Theme */}
                  <div className="p-6 overflow-y-auto max-h-[calc(85vh-120px)] bg-gradient-to-br from-green-50 to-emerald-50">
                    <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 gap-3">
                      {papers[selectedSubject].map((year, index) => {
                        const gradients = [
                          'from-green-500 to-emerald-500',
                          'from-emerald-500 to-teal-500',
                          'from-teal-500 to-cyan-500',
                          'from-lime-500 to-green-500',
                        ]
                        const gradient = gradients[index % gradients.length]
                        
                        const handleClick = (e: React.MouseEvent) => {
                          // Check limits for anonymous users
                          if (!user && !usageTracker.canViewPaper()) {
                            e.preventDefault()
                            setShowSignUpModal(true)
                            return
                          }
                          
                          // Increment counter for anonymous users
                          if (!user) {
                            usageTracker.incrementPaperView()
                          }
                        }
                        
                        return (
                          <Link
                            key={year}
                            href={`/past-papers/view?subject=${selectedSubject}&year=${year}`}
                            onClick={handleClick}
                            className="block group"
                          >
                            <div className={`relative bg-gradient-to-br ${gradient} rounded-xl p-4 hover:shadow-xl transition-all text-center overflow-hidden`}>
                              {/* Shine effect on hover */}
                              <div className="absolute inset-0 bg-white/0 group-hover:bg-white/20 transition-all"></div>
                              
                              <Calendar className="w-6 h-6 text-white mx-auto mb-2 relative z-10" />
                              <p className="font-bold text-lg text-white relative z-10">{year}</p>
                            </div>
                          </Link>
                        )
                      })}
                    </div>
                  </div>
                </div>
              </div>
            )}

            {/* Sign-Up Modal */}
            <SignUpModal
              isOpen={showSignUpModal}
              onClose={() => setShowSignUpModal(false)}
              usageStats={{
                papersViewed: usageTracker.getUsage().papersViewed
              }}
            />
          </div>

          <FeedbackButton page="past-papers" />
        </div>
      </ProtectedContent>
    </>
  )
}
